file(GLOB test_sources RELATIVE "${CMAKE_CURRENT_LIST_DIR}" "*.cpp")

file(COPY test.sh DESTINATION ${CMAKE_CURRENT_BINARY_DIR} FILE_PERMISSIONS OWNER_READ OWNER_EXECUTE)
file(COPY show_info.sh DESTINATION ${CMAKE_CURRENT_BINARY_DIR} FILE_PERMISSIONS OWNER_READ OWNER_EXECUTE)
file(COPY benchmark_docker.sh DESTINATION ${CMAKE_CURRENT_BINARY_DIR} FILE_PERMISSIONS OWNER_READ OWNER_EXECUTE)

foreach(test_source ${test_sources})
    string( REPLACE ".cpp" "" test_name ${test_source} )
    if((test_name STREQUAL "tpch_q6" OR test_name STREQUAL "tpch_q14" OR test_name STREQUAL "tpch_q3" OR test_name STREQUAL "tpch_q5" OR test_name STREQUAL "repack_test") AND NOT OpenFHE_FOUND)
        continue()
    endif()
    add_executable(${test_name} ${test_source})
    if(test_name STREQUAL "pruned_comparison_test" OR test_name STREQUAL "gapmsb_closed_loop_test")
        target_link_libraries(${test_name} hedb tfhe++ AES)
    elseif(test_name STREQUAL "tpch_q6" OR test_name STREQUAL "tpch_q14" OR test_name STREQUAL "tpch_q3" OR test_name STREQUAL "tpch_q5" OR test_name STREQUAL "repack_test")
        target_link_libraries(${test_name} hedb tfhe++ AES ${HEDB_OPENFHE_LIBRARIES})
        target_compile_options(${test_name} PRIVATE -Wno-deprecated-declarations)
    elseif(USE_PERF)
        add_compile_definitions(USE_PERF)
        target_link_libraries(${test_name} tfhe++ AES profiler)
    else()
        target_link_libraries(${test_name} tfhe++ AES)
    endif()
endforeach(test_source ${test_sources})

    add_compile_definitions(ENABLE_LSHTEST)
    target_link_libraries(INTorus gmp)
endif()
